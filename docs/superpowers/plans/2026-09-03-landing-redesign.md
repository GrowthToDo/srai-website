# SimpleRosterAI Landing Redesign Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rebuild the SimpleRosterAI home page as a Pain–Person–Promise landing page on a navy-and-saffron visual system with a new logo, restyle the inner pages through shared tokens, correct the hours claim, fix the founder photo, and ship an Indian-context launch video in the hero.

**Architecture:** Design tokens live in `CustomStyles.astro`; every existing component keeps its Tailwind class names so a token swap restyles the whole site. The home page becomes a composition of small single-purpose Astro widgets. The launch video is a fork of the sibling site's Remotion project, kept outside this repo, whose renders are copied into `public/videos/`.

**Tech Stack:** Astro 5.12, Tailwind 3.4, Svelte 5, `@fontsource-variable/manrope`, Python 3 + Pillow (icons, OG image), Remotion 4 (video, separate project), Node 22.

**Spec:** `docs/superpowers/specs/2026-09-03-landing-redesign-design.md`

```python
"""Generate favicon SVG/PNGs and app icons from the duty-wheel mark. Run: python scripts/build-icons.py"""
from pathlib import Path
from PIL import Image, ImageDraw

NAVY = (11, 31, 58)
SAFFRON = (245, 166, 35)
SVG = """<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 48 48"><rect width="48" height="48" rx="10" fill="#0B1F3A"/><circle cx="24" cy="24" r="15" fill="none" stroke="#F7F8FA" stroke-opacity="0.18" stroke-width="6"/><path d="M24 9 A15 15 0 0 1 37 16.5" fill="none" stroke="#F5A623" stroke-width="6" stroke-linecap="round"/><path d="M37 31.5 A15 15 0 0 1 24 39" fill="none" stroke="#F5A623" stroke-width="6" stroke-linecap="round" stroke-opacity="0.75"/><path d="M11 31.5 A15 15 0 0 1 11 16.5" fill="none" stroke="#F5A623" stroke-width="6" stroke-linecap="round" stroke-opacity="0.5"/><circle cx="24" cy="24" r="3.5" fill="#F7F8FA"/></svg>"""

def png(size: int, path: Path) -> None:
    s = size * 4  # supersample
    im = Image.new("RGBA", (s, s), (0, 0, 0, 0))
    d = ImageDraw.Draw(im)
    d.rounded_rectangle((0, 0, s, s), radius=s * 10 // 48, fill=NAVY)
    r = s * 15 // 48
    c = s // 2
    w = s * 6 // 48
    box = (c - r, c - r, c + r, c + r)
    d.arc(box, 0, 360, fill=(247, 248, 250, 46), width=w)
    d.arc(box, -90, -30, fill=SAFFRON, width=w)
    d.arc(box, 30, 90, fill=SAFFRON + (191,), width=w)
    d.arc(box, 150, 210, fill=SAFFRON + (128,), width=w)
    dot = s * 3.5 / 48
    d.ellipse((c - dot, c - dot, c + dot, c + dot), fill=(247, 248, 250))
    im = im.resize((size, size), Image.LANCZOS)
    path.parent.mkdir(parents=True, exist_ok=True)
    im.save(path, optimize=True)

for p in [Path("src/assets/favicons/favicon.svg"), Path("public/icon.svg")]:
    p.write_text(SVG, encoding="utf-8")
png(180, Path("src/assets/favicons/apple-touch-icon.png"))
png(180, Path("public/apple-touch-icon.png"))
png(192, Path("public/icon-192.png"))
png(512, Path("public/icon-512.png"))
ico = Image.open("public/icon-192.png").convert("RGBA")
ico.save("src/assets/favicons/favicon.ico", sizes=[(16, 16), (32, 32), (48, 48)])
ico.save("public/favicon.ico", sizes=[(16, 16), (32, 32), (48, 48)])
print("icons written")
```

<script>
  function initPinned() {
    const root = document.querySelector('[data-pinned-steps]');
    if (!root) return;
    const mq = window.matchMedia('(min-width: 1024px) and (prefers-reduced-motion: no-preference)');
    const items = Array.from(root.querySelectorAll<HTMLElement>('[data-rail-item]'));
    const cards = Array.from(root.querySelectorAll<HTMLElement>('[data-step-card]'));
    if (!items.length || !cards.length) return;
    const setActive = (i: number) => items.forEach((el, j) => (j === i ? el.setAttribute('data-active', '') : el.removeAttribute('data-active')));
    setActive(0);
    if (!mq.matches) return;
    const io = new IntersectionObserver(
      (entries) => {
        const best = entries.filter((e) => e.isIntersecting).sort((a, b) => b.intersectionRatio - a.intersectionRatio)[0];
        if (best) setActive(Number((best.target as HTMLElement).dataset.stepCard));
      },
      { rootMargin: '-35% 0px -35% 0px', threshold: [0.1, 0.5, 0.9] }
    );
    cards.forEach((c) => io.observe(c));
  }
  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', initPinned);
  else initPinned();
</script>

```ts
// Real data driven by the same engine the site demo uses. Nothing here is mocked.
import {
  NURSES, DAY_NAMES, DATASET, SHIFT_TYPES, SHIFT_LABEL, generateSchedule, checkSchedule, eligibleCandidates, designation,
  type Nurse, type DayIndex, type ShiftType, type Candidate,
} from './engine';

export type Tone = 'charge' | 'rn' | 'lpn' | 'cna'; // kept so ui.tsx's Chip works unchanged: charge=in-charge, rn=staff nurse, cna=assistant
export const tone = (n: Nurse): Tone => (n.inChargeQualified ? 'charge' : n.role === 'NA' ? 'cna' : 'rn');

const byId = new Map(NURSES.map((n) => [n.id, n]));
export const nurseById = (id: string): Nurse => byId.get(id)!;
export const shortName = (n: Nurse): string => {
  const [first, last] = n.name.split(' ');
  return last ? `${first} ${last[0]}.` : first;
};
export { designation, SHIFT_TYPES, SHIFT_LABEL };

const SCHEDULE = generateSchedule();
const forShift = (day: DayIndex, type: ShiftType): Nurse[] =>
  SCHEDULE.filter((a) => a.day === day && a.type === type).map((a) => nurseById(a.nurseId));

export interface DayRow { dayName: string; short: string; shifts: Record<ShiftType, Nurse[]> }
export const weekRows: DayRow[] = DAY_NAMES.map((dayName, d) => ({
  dayName,
  short: dayName.slice(0, 3),
  shifts: { morning: forShift(d as DayIndex, 'morning'), evening: forShift(d as DayIndex, 'evening'), night: forShift(d as DayIndex, 'night') },
}));

export interface Chip { nurse: Nurse; row: number; shift: ShiftType }
export const revealChips: Chip[] = SHIFT_TYPES.flatMap((shift) => weekRows.flatMap((r, i) => r.shifts[shift].map((nurse) => ({ nurse, row: i, shift }))));

const VIOLATIONS = checkSchedule(SCHEDULE);
const COVERAGE = new Set(['inchargeCoverage', 'snCoverage', 'naCoverage']);
const CHECK_DEFS: { label: string; rules: string[] }[] = [
  { label: '12-hour rest between duties', rules: ['rest12h'] },
  { label: 'One weekly off, no 7th day', rules: ['weeklyOff', 'maxConsecutive6'] },
  { label: 'In-charge on every shift', rules: ['inchargeCoverage'] },
  { label: 'Right skill mix on every shift', rules: ['snCoverage', 'naCoverage'] },
  { label: "Under the ward's hours cap", rules: ['maxHours48'] },
  { label: 'Approved leave respected', rules: ['approvedLeave'] },
];
export type CheckState = 'done' | 'warn' | 'bad';
export interface CheckItem { label: string; state: CheckState }
export const checklist: CheckItem[] = CHECK_DEFS.map(({ label, rules }) => {
  const vs = VIOLATIONS.filter((v) => rules.includes(v.rule));
  return { label, state: vs.length === 0 ? 'done' : vs.some((v) => COVERAGE.has(v.rule)) ? 'warn' : 'bad' };
});

// Absence scenario: a staff nurse drops from Thursday night; rank replacements.
const CO_DAY = 3 as DayIndex;
const CO_TYPE: ShiftType = 'night';
const assigned = forShift(CO_DAY, CO_TYPE);
const dropped = assigned.find((n) => n.role === 'SN' && !n.inChargeQualified) ?? assigned[0];
const after = SCHEDULE.filter((a) => !(a.nurseId === dropped.id && a.day === CO_DAY && a.type === CO_TYPE));
const all: Candidate[] = eligibleCandidates(after, { day: CO_DAY, type: CO_TYPE }, 'sn', DATASET);
const eligible = all.filter((c) => c.eligible && !c.already && c.nurse.id !== dropped.id).sort((a, b) => b.score - a.score);
export const callout = { dayName: DAY_NAMES[CO_DAY], shiftLabel: 'Thursday night shift', dropped, candidates: eligible.slice(0, 4), topId: eligible[0]?.nurse.id ?? null };
```
